aditya@lima-devops:~/devops-lab3$ cat ansible/deploy.yml
---
- name: Configure and deploy Lab 3 application
  hosts: webservers
  become: true

  tasks:
    - name: Install Nginx
      ansible.builtin.apt:
        name: nginx
        state: present
        update_cache: true

    - name: Deploy application
      ansible.builtin.copy:
        src: ../app/index.html
        dest: /var/www/html/index.html
        mode: '0644'

    - name: Start and enable Nginx
      ansible.builtin.systemd:
        name: nginx
        state: started
        enabled: true
